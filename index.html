Kiểm tra giới hạn Banlist dựa theo Tên for (const [cardName, totalCount] of Object.entries(nameCounts)) { const banInfo = banlistNameMap.get(cardName.toLowerCase().trim()); let maxAllowed = 3; let statusText = "Unlimited"; if (banInfo) { if (banInfo.status === 'Forbidden') maxAllowed = 0; else if (banInfo.status === 'Limited') maxAllowed = 1; else if (banInfo.status === 'Semi-Limited') maxAllowed = 2; statusText = banInfo.status; } if (totalCount > maxAllowed) { violations.push({ name: cardName, count: totalCount, maxAllowed: maxAllowed, status: statusText }); } } // 5.
